# Стартовый скрипт этапа 3: работа с VFS.
# Скрипт не зависит от содержимого VFS, поэтому подходит для любой из них.
pwd
ls
ls -a /
# Служебная команда: заменить VFS на VFS по умолчанию,
# файл VFS при этом перезаписывается
vfs-init
pwd
ls /
ls /home/user
# Ошибка: vfs-init не принимает аргументов, скрипт остановится
vfs-init extra
ls this-line-is-never-executed
